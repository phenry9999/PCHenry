Setting up resources on a page\Window and setting Styles by target type
	<Window.Resources>
		<SolidColorBrush x:Key="AvantiYellow" Color="#FFC40D" />
		<SolidColorBrush x:Key="AvantiRed" Color="#D9004A" />
		<SolidColorBrush x:Key="AvantiDarkGrey" Color="#707070" />
		<SolidColorBrush x:Key="AvantiLightGrey" Color="#F1F1F1" />

		<Style TargetType="TextBox">
			<Setter Property="Margin" Value="3" />
			<Setter Property="VerticalAlignment" Value="Center" />
			<Style.Triggers>
				<Trigger Property="Validation.HasError" Value="True">
					<Setter Property="Background" Value="Red" />
					<Setter Property="ToolTip" Value="{Binding RelativeSource={RelativeSource Self}, Path=(Validation.Errors)[0].ErrorContent}" />
				</Trigger>
			</Style.Triggers>
		</Style>
		<Style TargetType="TextBlock">
			<Setter Property="VerticalAlignment" Value="Center" />
		</Style>
		<Style TargetType="Label">
			<Setter Property="Margin" Value="3" />
			<Setter Property="VerticalAlignment" Value="Center" />
			<Style.Triggers>
				<Trigger Property="Validation.HasError" Value="True">
					<Setter Property="Background" Value="Red" />
					<Setter Property="ToolTip" Value="{Binding RelativeSource={RelativeSource Self}, Path=(Validation.Errors)[0].ErrorContent}" />
				</Trigger>
			</Style.Triggers>
		</Style>
